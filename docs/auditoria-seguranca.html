AUD-01 — Normalização de URI divergente e path traversal contornável

Severidade antes: Alta   Depois: Corrigido   Categoria: Roteamento / Path traversal   CWE/OWASP: CWE-22 / OWASP A01:2021

Onde (antes): src/Request.php:67-83; src/RouteMatcher.php:50-64
Onde (depois): src/UriNormalizer.php:10-60; src/Request.php:116-119; src/RouteMatcher.php:87-93

Por quê: Request decodificava com urldecode e só bloqueava os literais "/../", "../" e ".."; RouteMatcher tinha outra normalização, sem decodificação nem proteção de traversal. As duas normalizações podiam divergir. Double-encoding (%252e%252e%252f), backslashes e segmentos ".." não literais podiam escapar da checagem. Exemplo: GET /public/%252e%252e%252f%252e%252e%252fetc%252fpasswd.

Código ANTES

private function normalizeUri(string $uri): string
{
    $uri = urldecode($uri);
    $uri = str_replace('\\', '/', $uri);
    // basic traversal protection
    if (str_contains($uri, '/../') || str_starts_with($uri, '../') || $uri === '..') {
        $uri = '/';
    }
    $uri = preg_replace('#/+#', '/', $uri) ?? $uri;
    if ($uri === '') { $uri = '/'; }
    if ($uri !== '/' && str_ends_with($uri, '/')) { $uri = rtrim($uri, '/'); }
    return $uri;
}

final class UriNormalizer
{
    public static function normalize(string $uri): string
    {
        if (strlen($uri) > self::HARD_MAX_LENGTH) {
            throw new HttpException('URI muito longa', 414);
        }
        if (self::hasControlChars($uri)) {
            return '/';
        }
        $uri = self::decode($uri);
        if (self::hasControlChars($uri)) {
            return '/';
        }
        $uri = str_replace('\\', '/', $uri);
        $uri = preg_replace('#/+#', '/', $uri) ?? $uri;
        if ($uri === '') { return '/'; }
        if ($uri[0] !== '/') { $uri = '/' . $uri; }
        foreach (explode('/', $uri) as $segment) {
            if ($segment === '..') { return '/'; }
        }
        if ($uri !== '/' && str_ends_with($uri, '/')) {
            $uri = rtrim($uri, '/');
        }
        return $uri;
    }

    private static function decode(string $uri): string
    {
        for ($i = 0; $i < self::MAX_DECODE_PASSES; $i++) {
            $decoded = rawurldecode($uri);
            if ($decoded === $uri) { break; }
            $uri = $decoded;
        }
        return $uri;
    }
}

private function buildRegex(string $uri, array $wheres): string
{
    $parts = preg_split('/(\{[a-zA-Z_][a-zA-Z0-9_-]*\??\})/', $uri, -1,
        PREG_SPLIT_DELIM_CAPTURE | PREG_SPLIT_NO_EMPTY);
    $regex = '';
    $index = 0;
    foreach ($parts as $part) {
        if (preg_match('/^\{([a-zA-Z_][a-zA-Z0-9_-]*)(\?)?\}$/', $part, $m) === 1) {
            $pattern = $this->paramPattern($m[1], $wheres);
            // ...
            $regex .= '(?P<p' . $index . '>' . $pattern . ')';
            $index++;
        } else {
            $regex .= preg_quote($part, '#'); // escapa todo o literal
        }
    }
    return '#^' . $regex . '\z#uD';
}

private function resolve(mixed $middleware): mixed
{
    if (is_string($middleware)) {
        if ($this->container !== null && $this->container->has($middleware)) {
            $instance = $this->container->get($middleware);
            if (!is_object($instance)) { throw new \RuntimeException('...'); }

private function assertNoDuplicate(Route $route): void
{
